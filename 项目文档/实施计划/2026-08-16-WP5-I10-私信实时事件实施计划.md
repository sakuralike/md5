# WP5-I10 私信实时事件 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为一对一私信增加可断线补偿的 `message.created`、`conversation.read`、`unread.changed` 实时事件，并在 Web、远端分支和 Staging 多实例环境完成验证。

**Architecture:** MySQL 8 以每用户连续序号保存私信持久事件，Redis Pub/Sub 只发布无敏感内容的用户流唤醒信号，SSE 使用 `Last-Event-ID` 从数据库补偿。Web 使用独立 Pinia Store 与流客户端维护总未读、会话未读和对方已读序号；页面收到事件后合并刷新授权 API，不从事件负载读取私信正文。

**Tech Stack:** Python 3.12、FastAPI、SQLAlchemy、Alembic、MySQL 8/SQLite、Redis 5+、Vue 3、Pinia、TypeScript 严格模式、Shadcn-Vue、Tailwind CSS 3、pnpm、pytest、Vitest、Playwright、Docker Compose。

```python
@dataclass(frozen=True)
class DirectEventDraft:
    recipient_id: str
    event_type: CommunityDirectEventType
    conversation_id: str
    actor_id: str
    message_id: str | None
    payload: dict[str, object]


def ensure_stream_position(db: Session, user_id: str) -> None:
    values = {"user_id": user_id, "last_sequence": 0, "created_at": utc_now(), "updated_at": utc_now()}
    dialect = db.bind.dialect.name
    if dialect == "mysql":
        db.execute(mysql_insert(CommunityDirectStreamPosition).values(**values).on_duplicate_key_update(user_id=user_id))
    elif dialect == "sqlite":
        db.execute(sqlite_insert(CommunityDirectStreamPosition).values(**values).on_conflict_do_nothing(index_elements=["user_id"]))
    else:
        if db.get(CommunityDirectStreamPosition, user_id) is None:
            db.add(CommunityDirectStreamPosition(**values))
            db.flush()


def append_direct_events(db: Session, drafts: Sequence[DirectEventDraft]) -> list[CommunityDirectEvent]:
    recipients = sorted({draft.recipient_id for draft in drafts})
    for recipient_id in recipients:
        ensure_stream_position(db, recipient_id)
    positions = {
        row.user_id: row
        for row in db.scalars(
            select(CommunityDirectStreamPosition)
            .where(CommunityDirectStreamPosition.user_id.in_(recipients))
            .order_by(CommunityDirectStreamPosition.user_id)
            .with_for_update()
        )
    }
    created: list[CommunityDirectEvent] = []
    for draft in drafts:
        position = positions[draft.recipient_id]
        position.last_sequence += 1
        position.updated_at = utc_now()
        event = CommunityDirectEvent(id=new_id(), sequence=position.last_sequence, **asdict(draft), created_at=utc_now())
        db.add(event)
        created.append(event)
    db.flush()
    return created
```

```python
def direct_stream_channel(user_id: str) -> str:
    return f"community:direct-stream:user:{user_id}"


def publish_direct_message_wakeups(redis_url: str, user_sequences: Mapping[str, int]) -> None:
    try:
        client = redis.Redis.from_url(redis_url, decode_responses=True)
        for user_id, sequence in sorted(user_sequences.items()):
            client.publish(direct_stream_channel(user_id), json.dumps({"max_sequence": sequence}, separators=(",", ":")))
    except redis.RedisError:
        logger.warning("Direct message Redis wakeup failed", exc_info=True)


class DirectStreamWakeSubscription:
    async def wait(self, timeout_seconds: float) -> bool:
        message = await asyncio.wait_for(self._pubsub.get_message(ignore_subscribe_messages=True, timeout=timeout_seconds), timeout=timeout_seconds + 0.1)
        return message is not None
```

路由必须在 `_mutate_with_idempotency()` 已提交后查询会话双方最新用户流序号并发布；缓存命中或重复唤醒允许存在，不得回滚业务事务。

```python
async def direct_message_event_stream(request: Request, recipient_id: str, raw_last_event_id: str | None) -> AsyncIterator[str]:
    cursor = load_cursor_snapshot(recipient_id, raw_last_event_id)
    yield encode_sse("ready", cursor.latest_sequence, DirectStreamReady(event_id=cursor.latest_sequence, total_unread_count=cursor.total_unread_count, reset_required=cursor.reset_required))
    after_sequence = cursor.after_sequence
    subscription = await DirectStreamWakeSubscription.open(settings.redis_url, recipient_id)
    try:
        while not await request.is_disconnected():
            batch = load_event_batch(recipient_id, after_sequence, limit=100)
            for event in batch.items:
                yield encode_sse(event.event_type.value, event.sequence, event_payload(event))
                after_sequence = event.sequence
            if batch.items:
                continue
            if subscription is None:
                await asyncio.sleep(1.0)
            else:
                woke = await subscription.wait(15.0)
                if not woke:
                    yield ": heartbeat\n\n"
    finally:
        if subscription is not None:
            await subscription.close()
```

`load_cursor_snapshot()` 和 `load_event_batch()` 每次使用 `with session_factory() as db:`，离开函数前关闭 Session；`GET /community/direct-messages/stream` 继续使用 function-scope 认证并设置 `Cache-Control: no-cache, no-transform`、`X-Accel-Buffering: no`。

```typescript
export const useCommunityDirectMessagesStore = defineStore('communityDirectMessages', () => {
  const latestEventId = ref(0)
  const totalUnreadCount = ref(0)
  const conversationUnread = reactive<Record<string, number>>({})
  const counterpartReadSequence = reactive<Record<string, number>>({})
```
